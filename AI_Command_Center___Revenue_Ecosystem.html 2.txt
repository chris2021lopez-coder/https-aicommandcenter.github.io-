    function resolveApproval(id, approved) {

      if (approved) {
        const targetDept = state.departments.find(d => d.name === item.dept) || state.departments[0];
        if (state.isDemo) targetDept.demoRevenue += item.amount;
        else targetDept.liveRevenue += item.amount;
        targetDept.tasksCompleted += 1;

        const timeStr = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
        state.ledger.unshift({
          id: 'tx-' + Date.now(),
          time: timeStr,
          dept: item.dept,
          desc: `Approved: ${item.title}`,
          type: state.isDemo ? 'DEMO' : 'LIVE',
          amount: item.amount

        spawnPulseForDept(targetDept.id);
        showToast('ACTION APPROVED & EXECUTED', `Settled +${formatCurrency(item.amount)} to ${item.dept}`, 'emerald');

        showToast('PROPOSAL REJECTED', `Blocked: ${item.title}`, 'amber');

      saveState();
      renderAll();
    }

    function drawNeuralNetwork() {
      ctx.clearRect(0, 0, canvasW, canvasH);

      const cx = canvasW / 2;
      const cy = canvasH / 2 + 8;
      const orbRadius = Math.min(canvasW, canvasH) * 0.17;

      // 1. Central Ambient Radial Glow
      const grad = ctx.createRadialGradient(cx, cy, 4, cx, cy, orbRadius * 2.1);
      grad.addColorStop(0, 'rgba(0, 240, 255, 0.22)');
      grad.addColorStop(0.5, 'rgba(16, 185, 129, 0.06)');
      grad.addColorStop(1, 'rgba(9, 11, 16, 0)');
      ctx.fillStyle = grad;
      ctx.beginPath();
      ctx.arc(cx, cy, orbRadius * 2.1, 0, Math.PI * 2);
      ctx.fill();

      // 2. Compute Orbiting Node Coordinates & Draw Synaptic Conduits
      const spreadX = Math.min(canvasW * 0.88, 680);
      const spreadY = Math.min(canvasH * 0.85, 340);
      renderedNodeHitboxes = [];

      nodeLayout.forEach((node, index) => {
        const dept = state.departments.find(d => d.id === node.id);
        const floatY = Math.sin(orbRotation * 2 + index) * 4;
        const nx = cx + node.rx * spreadX;
        const ny = cy + node.ry * spreadY + floatY;

        // Conduit line from center orb to node
        ctx.strokeStyle = dept && dept.active ? `${node.color}44` : 'rgba(100,116,139,0.18)';
        ctx.lineWidth = 1.4;
        ctx.beginPath();
        ctx.moveTo(cx, cy);
        ctx.lineTo(nx, ny);
        ctx.stroke();

        // Glowing Orb Satellite
        const satGrad = ctx.createRadialGradient(nx, ny, 1, nx, ny, 18);
        satGrad.addColorStop(0, node.color);
        satGrad.addColorStop(1, 'rgba(0,0,0,0)');
        ctx.fillStyle = satGrad;
        ctx.beginPath();
        ctx.arc(nx, ny, 18, 0, Math.PI * 2);
        ctx.fill();

        ctx.fillStyle = dept && dept.active ? node.color : '#475569';
        ctx.beginPath();
        ctx.arc(nx, ny, 5.5, 0, Math.PI * 2);
        ctx.fill();

        // Pill Label Badge (Matches Screenshot 2 aesthetic)
        const pillText = node.label;
        ctx.font = '600 10px "JetBrains Mono", monospace';
        const textW = ctx.measureText(pillText).width;
        const pillW = textW + 28;
        const pillH = 24;
        const pillX = nx - pillW / 2;
        const pillY = ny - 32;

        // Pill Background
        ctx.fillStyle = 'rgba(11, 15, 25, 0.92)';
        ctx.strokeStyle = dept && dept.active ? `${node.color}88` : '#334155';
        ctx.lineWidth = 1;
        roundRect(ctx, pillX, pillY, pillW, pillH, 12);
        ctx.fill();
        ctx.stroke();

        // Status Dot inside pill
        ctx.fillStyle = dept && dept.active ? node.color : '#64748b';
        ctx.beginPath();
        ctx.arc(pillX + 11, pillY + pillH / 2, 3, 0, Math.PI * 2);
        ctx.fill();

        // Pill Text
        ctx.fillStyle = '#e2e8f0';
        ctx.textAlign = 'left';
        ctx.textBaseline = 'middle';
        ctx.fillText(pillText, pillX + 19, pillY + pillH / 2 + 0.5);

        renderedNodeHitboxes.push({
          id: node.id,
          x: Math.min(pillX, nx - 18),
          y: pillY,
          w: Math.max(pillW, 36),
          h: 52
        });
      });

      // 3. Draw Data Pulses traveling along conduits
      activePulses = activePulses.filter(p => p.progress < 1);
      activePulses.forEach(p => {
        p.progress += p.speed;
        const node = nodeLayout.find(n => n.id === p.deptId);
        if (!node) return;
        const nx = cx + node.rx * spreadX;
        const ny = cy + node.ry * spreadY;
        const px = cx + (nx - cx) * p.progress;
        const py = cy + (ny - cy) * p.progress;

        ctx.fillStyle = p.color;
        ctx.shadowColor = p.color;
        ctx.shadowBlur = 12;
        ctx.beginPath();
        ctx.arc(px, py, 4, 0, Math.PI * 2);
        ctx.fill();
        ctx.shadowBlur = 0;
      });

      // 4. Draw Rotating 3D Wireframe Geodesic Core Orb
      orbRotation += 0.008;
      const cosA = Math.cos(orbRotation);
      const sinA = Math.sin(orbRotation);

      const projected = sphereVertices.map(v => {
        const x1 = v.x * cosA - v.z * sinA;
        const z1 = v.z * cosA + v.x * sinA;
        return {
          x: cx + x1 * orbRadius,
          y: cy + v.y * orbRadius,
          z: z1
        };
      });

      ctx.strokeStyle = 'rgba(0, 240, 255, 0.35)';
      ctx.lineWidth = 0.9;
      for (let i = 0; i < projected.length; i++) {
        for (let j = i + 1; j < projected.length; j++) {
          const dx = projected[i].x - projected[j].x;
          const dy = projected[i].y - projected[j].y;
          const dist = Math.sqrt(dx * dx + dy * dy);
          if (dist < orbRadius * 0.56) {
            ctx.beginPath();
            ctx.moveTo(projected[i].x, projected[i].y);
            ctx.lineTo(projected[j].x, projected[j].y);
            ctx.stroke();
          }
        }
      }

      projected.forEach(pt => {
        ctx.fillStyle = pt.z > 0 ? '#00f0ff' : 'rgba(0, 240, 255, 0.25)';
        ctx.beginPath();
        ctx.arc(pt.x, pt.y, pt.z > 0 ? 2 : 1.2, 0, Math.PI * 2);
        ctx.fill();
      });

      // Central SUPPORT / CORE Badge inside the wireframe sphere
      ctx.font = '700 10px "JetBrains Mono", monospace';
      ctx.fillStyle = '#a5f3fc';
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText('SUPPORT • CORE', cx, cy);

      requestAnimationFrame(drawNeuralNetwork);
    }

    function roundRect(ctx, x, y, width, height, radius) {
      ctx.beginPath();
      ctx.moveTo(x + radius, y);
      ctx.lineTo(x + width - radius, y);
      ctx.quadraticCurveTo(x + width, y, x + width, y + radius);
      ctx.lineTo(x + width, y + height - radius);
      ctx.quadraticCurveTo(x + width, y + height, x + width - radius, y + height);
      ctx.lineTo(x + radius, y + height);
      ctx.quadraticCurveTo(x, y + height, x, y + height - radius);
      ctx.lineTo(x, y + radius);
      ctx.quadraticCurveTo(x, y, x + radius, y);
      ctx.closePath();
    }

    // Canvas Click / Tap Interaction
    canvas.addEventListener('click', (e) => {
      const rect = canvas.getBoundingClientRect();
      const clickX = e.clientX - rect.left;
      const clickY = e.clientY - rect.top;

      for (const box of renderedNodeHitboxes) {
        if (clickX >= box.x && clickX <= box.x + box.w && clickY >= box.y && clickY <= box.y + box.h) {
          runSingleAgentCycle(box.id);
          return;
        }
      }

      // Clicking center orb triggers full pulse
      const cx = canvasW / 2;
      const cy = canvasH / 2 + 8;
      if (Math.hypot(clickX - cx, clickY - cy) < 65) {
        triggerNetworkPulse();
      }
    });

    // Periodic ambient pulses so the network feels alive
    setInterval(() => {
      if (state.activeTab === 'home' && state.visualizerMode === 'neural') {
        const active = state.departments.filter(d => d.active);
        if (active.length > 0) {
          const rand = active[Math.floor(Math.random() * active.length)];
          spawnPulseForDept(rand.id);
        }
      }
    }, 1600);

    window.addEventListener('resize', () => {
      if (state.visualizerMode === 'neural') resizeNeuralCanvas();
    });

    // ==================== INITIALIZATION ====================
    window.onload = function () {
      renderAll();
      setVisualizerMode(state.visualizerMode || 'neural');
      switchTab(state.activeTab || 'home');
      resizeNeuralCanvas();
      drawNeuralNetwork();
    };
